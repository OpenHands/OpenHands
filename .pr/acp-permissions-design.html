+import { useEffect, useState } from "react";
+import { useTranslation } from "react-i18next";
+import type { AppConversation } from "#/api/conversation-service/agent-server-conversation-service.types";
+import { I18nKey } from "#/i18n/declaration";
+import { useAcpPermissions } from "#/hooks/query/use-acp-permissions";
+
+export function AcpPermissionPanel({
+  conversation,
+}: {
+  conversation: AppConversation | null | undefined;
+}) {
+  const { t } = useTranslation();
+  const { pending, decision, scope, enabled } = useAcpPermissions(conversation);
+  const [now, setNow] = useState(Date.now);
+  useEffect(() => {
+    if (!enabled) return undefined;
+    const timer = setInterval(() => setNow(Date.now()), 250);
+    return () => clearInterval(timer);
+  }, [enabled]);
+  if (!enabled) return null;
+  const error =
+    pending.isError ||
+    (decision.isError && decision.variables?.requestScope === scope);
+  return (
+    <section
+      aria-label={t(I18nKey.ACP_PERMISSION_REQUIRED)}
+      className="flex flex-col gap-2"
+    >
+      {error && (
+        <div role="alert">
+          {t(I18nKey.ERROR$GENERIC)}{" "}
+          <button
+            type="button"
+            onClick={() => {
+              decision.reset();
+              void pending.refetch();
+            }}
+          >
+            {t(I18nKey.BUTTON$REFRESH)}
+          </button>
+        </div>
+      )}
+      {pending.data?.items.map((request) => {
+        const expires = Date.parse(request.expires_at);
+        const expired = !Number.isFinite(expires) || expires <= now;
+        const busy =
+          decision.isPending && decision.variables?.requestScope === scope;
+        const decide = (optionId: string | null) =>
+          decision.mutate({
+            requestId: request.request_id,
+            optionId,
+            requestScope: scope,
+          });
+        return (
+          <article key={request.request_id} className="rounded border p-3">
+            <h2>{t(I18nKey.ACP_PERMISSION_REQUIRED)}</h2>
+            <p>{request.title}</p>
+            <pre className="max-h-64 overflow-auto whitespace-pre-wrap break-words">
+              {request.details}
+            </pre>
+            {request.details_truncated && (
+              <p role="alert">{t(I18nKey.ACP_PERMISSION_INCOMPLETE)}</p>
+            )}
+            <div className="flex flex-wrap gap-2">
+              {request.options.map((option) => (
+                <button
+                  key={option.id}
+                  type="button"
+                  disabled={
+                    busy ||
+                    expired ||
+                    pending.isError ||
+                    (request.details_truncated &&
+                      option.kind.startsWith("allow"))
+                  }
+                  onClick={() => decide(option.id)}
+                >
+                  {option.name}
+                </button>
+              ))}
+              <button
+                type="button"
+                disabled={busy || expired}
+                onClick={() => decide(null)}
+              >
+                {t(I18nKey.BUTTON$CANCEL)}
+              </button>
+            </div>
+          </article>
+        );
+      })}
+    </section>
+  );
+}

+import { useMutation, useQuery, useQueryClient } from "@tanstack/react-query";
+import { useMemo } from "react";
+import type { AppConversation } from "#/api/conversation-service/agent-server-conversation-service.types";
+import { AcpPermissionsService } from "#/api/acp-service/acp-permissions.api";
+import { useActiveBackend } from "#/contexts/active-backend-context";
+import { ACP_PERMISSION_QUERY_KEYS } from "./query-keys";
+
+const PERMISSION_POLL_MS = 1000;
+
+export function useAcpPermissions(
+  conversation: AppConversation | null | undefined,
+) {
+  const { backend, orgId } = useActiveBackend();
+  const queryClient = useQueryClient();
+  const authorization = useMemo(
+    () => ({
+      key: conversation?.session_api_key,
+      generation: crypto.randomUUID(),
+    }),
+    [conversation?.session_api_key],
+  );
+  const queryKey = ACP_PERMISSION_QUERY_KEYS.pending(
+    backend.id,
+    backend.connectionRevision,
+    orgId,
+    conversation?.id,
+    conversation?.conversation_url,
+    authorization.generation,
+  );
+  const scope = JSON.stringify(queryKey);
+  const enabled =
+    !!conversation?.id &&
+    conversation.agent_kind === "acp" &&
+    (backend.kind !== "cloud" ||
+      (!!conversation.conversation_url &&
+        !!conversation.session_api_key &&
+        conversation.sandbox_status === "RUNNING"));
+  const target = {
+    id: conversation?.id ?? "",
+    conversationUrl: conversation?.conversation_url,
+    sessionApiKey: authorization.key,
+  };
+  // Credentials stay in the transport closure. Opaque authorization generation,
+  // backend revision and runtime URL isolate every replacement without caching keys.
+  // eslint-disable-next-line @tanstack/query/exhaustive-deps
+  const pending = useQuery({
+    queryKey,
+    queryFn: () => AcpPermissionsService.list(target),
+    enabled,
+    retry: false,
+    refetchInterval: PERMISSION_POLL_MS,
+    refetchIntervalInBackground: false,
+    staleTime: 0,
+    gcTime: 0,
+  });
+  const decision = useMutation({
+    mutationFn: ({
+      requestId,
+      optionId,
+      requestScope,
+    }: {
+      requestId: string;
+      optionId: string | null;
+      requestScope: string;
+    }) => {
+      if (!enabled || requestScope !== scope)
+        throw new Error("Permission conversation changed");
+      return AcpPermissionsService.decide(target, requestId, {
+        option_id: optionId,
+      });
+    },
+    retry: false,
+    onSettled: () => queryClient.invalidateQueries({ queryKey, exact: true }),
+  });
+  return { pending, decision, scope, enabled };
+}
