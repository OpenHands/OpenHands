GHSA-55q2-fjhq-7xh7: dynamic reproduction

The same JSDOM script was run after npm ci on main and on the patched branch. It sanitizes in place while a hook removes the parent of an unsafe image.

Before — DOMPurify 3.4.12

{
  "version": "3.4.12",
  "retainedOnload": "window.__xss = true",
  "returnedHtml": "<div>safe</div>"
}

After — DOMPurify 3.4.14

{
  "version": "3.4.14",
  "retainedOnload": null,
  "returnedHtml": "<div>safe</div>"
}

Sanitization script

const { JSDOM } = require("jsdom");
const createDOMPurify = require("dompurify");
const window = new JSDOM("<!doctype html>").window;
const DOMPurify = createDOMPurify(window);
const root = window.document.createElement("div");
root.innerHTML =
  '<footer><img src="x" onload="window.__xss = true"></footer><div>safe</div>';
const image = root.querySelector("img");

DOMPurify.setConfig({
  ALLOWED_TAGS: ["div", "#text", "footer"],
  IN_PLACE: true,
});
DOMPurify.addHook("uponSanitizeElement", (node) => {
  if (node.tagName === "FOOTER") node.remove();
});
DOMPurify.sanitize(root);
